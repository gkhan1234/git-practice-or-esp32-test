import time

# Pin assignments on ESP32 DevKit
led_alert = Pin(18, Pin.OUT)
led_heartbeat = Pin(19, Pin.OUT)
night_light = Pin(2, Pin.OUT)
ldr = ADC(Pin(34))
ldr.atten(ADC.ATTN_11DB)

trig = Pin(5, Pin.OUT)
echo = Pin(18, Pin.IN)

print("The Micro Lab Live REPL Started")

while True:
    # 1. Read Light Sensor
    val = ldr.read()
    if val < 1200:
        night_light.value(1) # Auto night light ON
    else:
        night_light.value(0)
        
    # 2. Heartbeat flash
    led_heartbeat.value(not led_heartbeat.value())
    time.sleep(0.5)
